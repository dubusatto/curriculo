% --- Pacotes Necessários ---
\usepackage{xcolor}
\usepackage{titlesec} % Pacote mais robusto e padrão para títulos

% --- Definição das Cores ---
\definecolor{myblue}{RGB}{0, 74, 145}  % Seu azul #004a91
\definecolor{mygray}{RGB}{51, 51, 51}  % Cinza escuro #333

% --- Aplicação dos Estilos ---

% Estilo para # (H1 / \section)
\titleformat{\section}
  {\normalfont\bfseries\Huge\color{myblue}\raggedright} % \Huge é o maior tamanho
  {} 
  {0em}
  {} 

% Estilo para ## (H2 / \subsection)
\titleformat{\subsection}
  {\normalfont\bfseries\Large\color{mygray}\raggedright} % \Large é um bom tamanho
  {}
  {0em}
  {}

% Bônus: Remove a numeração (1. Resumo, 2. Habilidades)
\setcounter{secnumdepth}{0}

% Ajusta os espaços para o PDF não ficar muito "espalhado"
\titlespacing*{\section}{0pt}{*2}{*1}
\titlespacing*{\subsection}{0pt}{*2}{*1}
